SQLite Schema Definition (DDL)

            
Exact table schemas extracted from data/farmart.db

          

          
        

        
CREATE TABLE categories (
  id          TEXT PRIMARY KEY,
  name        TEXT NOT NULL,
  sort_order  INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE products (
  id                    TEXT PRIMARY KEY,
  name                  TEXT NOT NULL,
  brand                 TEXT NOT NULL,
  category_id           TEXT NOT NULL REFERENCES categories(id),
  price                 REAL NOT NULL,
  original_price        REAL NOT NULL,
  discount_percent      INTEGER NOT NULL DEFAULT 0,
  rating                REAL NOT NULL DEFAULT 0,
  review_count          INTEGER NOT NULL DEFAULT 0,
  unit                  TEXT NOT NULL,
  sku                   TEXT NOT NULL UNIQUE,
  stock_count           INTEGER NOT NULL DEFAULT 0,
  sold_count            INTEGER NOT NULL DEFAULT 0,
  total_stock           INTEGER NOT NULL DEFAULT 0,
  is_best_seller        INTEGER NOT NULL DEFAULT 0,
  is_top_saver          INTEGER NOT NULL DEFAULT 0,
  is_organic            INTEGER NOT NULL DEFAULT 0,
  description           TEXT NOT NULL,
  features_json         TEXT NOT NULL DEFAULT '[]',
  nutrition_json        TEXT NOT NULL DEFAULT '{}',
  storage_instructions  TEXT,
  origin                TEXT,
  shelf_life            TEXT
);
CREATE INDEX idx_products_category ON products(category_id);

CREATE TABLE reviews (
  id          TEXT PRIMARY KEY,
  product_id  TEXT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  author      TEXT NOT NULL,
  rating      INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  date        TEXT NOT NULL,
  title       TEXT,
  comment     TEXT NOT NULL,
  verified    INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX idx_reviews_product ON reviews(product_id);

CREATE TABLE brands (
  id          TEXT PRIMARY KEY,
  name        TEXT NOT NULL,
  headline    TEXT NOT NULL,
  background  TEXT NOT NULL,
  is_featured INTEGER NOT NULL DEFAULT 0,
  sort_order  INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE banners (
  id          TEXT PRIMARY KEY,
  placement   TEXT NOT NULL,
  title       TEXT NOT NULL,
  subtitle    TEXT,
  cta_label   TEXT NOT NULL,
  href        TEXT,
  is_active   INTEGER NOT NULL DEFAULT 1,
  sort_order  INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE deal_campaigns (
  id            TEXT PRIMARY KEY,
  title         TEXT NOT NULL,
  ends_at       TEXT NOT NULL,
  cycle_seconds INTEGER NOT NULL DEFAULT 86400
);

CREATE TABLE deal_items (
  campaign_id TEXT NOT NULL REFERENCES deal_campaigns(id) ON DELETE CASCADE,
  product_id  TEXT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  sort_order  INTEGER NOT NULL DEFAULT 0,
  is_featured INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (campaign_id, product_id)
);
